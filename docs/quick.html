The whole workflow

local
    smt: SIMPLE_SMT
    x, y, zero, one: SMT_EXPR
do
    create smt.make
    x := smt.int_const ("x"); y := smt.int_const ("y")
    zero := smt.int_value (0); one := smt.int_value (1)

    -- PROVE: valid for all values?
    smt.prove (x.at_least (zero).entails (x.plus (one).greater (zero)))   -- True
    smt.prove (x.less (y).entails (y.greater (x)))                        -- True

    -- REFUTE: counter-model if not valid
    smt.refutation (x.greater (zero))                                     -- "x -> 0"

    -- SOLVE: assert constraints, check, read model
    smt.assume (x.greater (smt.int_value (5)))
    smt.assume (x.less (smt.int_value (8)))
    if smt.is_satisfiable then print (smt.model_value (x)) end            -- 6 or 7
end

The prove pattern

To prove a conjecture valid, assert its negation and check for unsat. That is what prove does; understanding it explains everything else:

-- "prove P" succeeds iff (not P) is unsatisfiable, i.e. no counterexample exists.
-- "refutation P" returns that counterexample when one exists.
